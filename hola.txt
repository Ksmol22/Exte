-- Datos reales en todo el CDB (lo que el backup tendrá que leer)
SELECT ROUND(SUM(bytes)/1024/1024/1024,1) gb_datos FROM cdb_segments;

-- Archives generados por día (últimos 7 días)
SELECT TRUNC(completion_time) dia,
       ROUND(SUM(blocks*block_size)/1024/1024/1024,1) gb_archives
FROM v$archived_log
WHERE dest_id = 1 AND completion_time > SYSDATE-7
GROUP BY TRUNC(completion_time) ORDER BY 1;

-- Tamaño actual de la FRA
